// ── derived-tab-id ───────────────────────────────────────────────────────────
// cm-assessment-workspace: "Audit Trail" became "audit", so the placeholder
// advertised "building the audit functionality" — a tab that could not name
// itself, and renaming any label silently repointed its tab.
export function DerivedTabIds() {
  return (
    <div>
      {['Overview', 'Participants', 'Ratings', 'Calibration', 'Audit Trail'].map(tab => {
        const id = tab.toLowerCase().split(' ')[0]
        return <button key={tab} onClick={() => setCampaignTab(id)}>{tab}</button>
      })}
    </div>
  )
}

// ── silent-status-check ──────────────────────────────────────────────────────
// mobility-center, seventeen times: the API answers 200 with {status: 0} for a
// refused save, and that branch did nothing at all. The drawer stayed open and
// the screen said nothing, so a rejected save looked exactly like a slow one.
export async function SilentSave() {
  try {
    const res = await mobilityService.createJob(payload)
    if (res.status === 1) {
      fetchListData()
      loadFiltersAndOverview()
    }
  } catch (err: any) {
    alert(err.message || 'Failed to create job posting.')
  }
}
